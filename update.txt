    """
    Khởi động `ollama serve` TRONG container (không có systemd), chạy nền,
    rồi chờ tới khi API trả lời. Gọi 1 lần trước khi chạy benchmark.
    """
    import os
    import shutil
    import subprocess
    import time
    import urllib.request

    api = f"http://{host}"

    def is_up():
        try:
            urllib.request.urlopen(f"{api}/api/version", timeout=2).read()
            return True
        except Exception:
            return False

    def cgroup_limit_gb():
        for p in ("/sys/fs/cgroup/memory.max",
                  "/sys/fs/cgroup/memory/memory.limit_in_bytes"):
            try:
                v = open(p).read().strip()
                if v == "max":
                    return None
                n = int(v)
                return n / 1024**3 if n < 1 << 60 else None
            except Exception:
                continue
        return None

    print("=" * 60, flush=True)
    print("START OLLAMA SERVER", flush=True)
    print("=" * 60, flush=True)

    # ---- Giới hạn RAM thật của container ----
    lim = cgroup_limit_gb()
    if lim:
        print(f"[Info] Giới hạn RAM của container (cgroup): {lim:.1f} GB", flush=True)
    else:
        print("[Info] Không thấy giới hạn RAM cgroup (hoặc là 'max')", flush=True)

    ollama_bin = shutil.which("ollama") or "/usr/bin/ollama"
    if not os.path.exists(ollama_bin):
        print("[Error] Chưa có binary ollama. Chạy lại bước cài đặt trước.", flush=True)
        return False

    if is_up():
        print("[OK] Server đã chạy sẵn.", flush=True)
    else:
        env = os.environ.copy()
        env.update({
            "OLLAMA_HOST": host,
            "OLLAMA_MODELS": models_dir,
            "OLLAMA_NUM_PARALLEL": "8",
            "OLLAMA_MAX_LOADED_MODELS": "1",
            "OLLAMA_FLASH_ATTENTION": "1",
            "OLLAMA_KEEP_ALIVE": "10m",
            # "OLLAMA_SCHED_SPREAD": "1",   # bỏ comment nếu có nhiều GPU
        })

        log = open(log_path, "w")
        proc = subprocess.Popen(
            [ollama_bin, "serve"],
            env=env,
            stdout=log,
            stderr=subprocess.STDOUT,
            start_new_session=True,   # sống tiếp dù cell kết thúc
        )
        print(f"[System] Đã chạy ollama serve (pid {proc.pid}), log: {log_path}", flush=True)

        t0 = time.time()
        while time.time() - t0 < wait_s:
            if proc.poll() is not None:
                print(f"[Error] Server thoát sớm (exit {proc.returncode}). Log:", flush=True)
                print(open(log_path).read()[-3000:], flush=True)
                return False
            if is_up():
                break
            time.sleep(1)
        else:
            print(f"[Error] Quá {wait_s}s mà server chưa trả lời. Log:", flush=True)
            print(open(log_path).read()[-3000:], flush=True)
            return False

        print(f"[OK] Server sẵn sàng sau {time.time() - t0:.1f}s", flush=True)

    # ---- Có thấy GPU không? ----
    try:
        time.sleep(1)
        txt = open(log_path).read()
        gpu_lines = [l for l in txt.splitlines()
                     if any(k in l.lower() for k in ("cuda", "gpu", "vram", "library="))]
        print("\n[GPU] Dòng liên quan trong log:", flush=True)
        for l in gpu_lines[-8:]:
            print("  ", l, flush=True)
        if not gpu_lines:
            print("   (không có dòng nào, xem log đầy đủ)", flush=True)
    except Exception:
        pass

    # ---- Danh sách model ----
    print("\n[Models]", flush=True)
    env2 = os.environ.copy()
    env2["OLLAMA_HOST"] = host
    subprocess.run([ollama_bin, "list"], env=env2)

    if pull_model:
        print(f"\n[System] ollama pull {pull_model} ...", flush=True)
        subprocess.run([ollama_bin, "pull", pull_model], env=env2)

    return True
